namespace ConsoleApp1
{
    internal class Program
    {
        static void Main(string[] args)
        {
            string[] peliculas = new string[5];

            for (int i = 0; i < 5; i++)
            {
                Console.Write("Ingresa una de tus peliculas favoritas: ");
                string lectura = Console.ReadLine();
                peliculas[i] = lectura;
            }

            Console.WriteLine("Tus peliculas favoritas son:");

            foreach (string peli in peliculas)
            {
                Console.WriteLine(peli);
            }



        }
    }
}
