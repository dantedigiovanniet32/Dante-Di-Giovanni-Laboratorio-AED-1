namespace ConsoleApp1
{
    internal class Program
    {
        static void Main(string[] args)
        {

            int[] numeros = new int[8];
            int mayor = 0;
            int posicion = 0;

            for (int i = 0; i < 8; i++)
            {
                Console.Write("Ingresa un numero: ");
                numeros[i] = Convert.ToInt32(Console.ReadLine());
                
            }


            mayor = numeros[0];
            posicion = 0;

            for (int i = 1; i < numeros.Length; i++)
            {
                if (numeros[i] > mayor)
                {
                    mayor = numeros[i];
                    posicion = i;
                }
            }

            Console.WriteLine("Numero mas grande: " + mayor);
            Console.WriteLine("Esta en la posicion: " + posicion);

        }
    }
}
