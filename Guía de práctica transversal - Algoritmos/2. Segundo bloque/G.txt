namespace ConsoleApp1
{
    internal class Program
    {
        static void Main(string[] args)
        {
            int contador = 0;

            for (int numero = 1; numero <= 100; numero++)
            {
                if (numero % 2 == 0)
                {
                    contador++;
                }
            }

            Console.WriteLine("Hay " + contador + " numeros pares");

        }
    }
}
