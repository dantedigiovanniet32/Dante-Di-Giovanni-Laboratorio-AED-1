namespace ConsoleApp1
{
    internal class Program
    {
        static void Main(string[] args)
        {
            for (int numero = 100; numero <= 200; numero++)
            {
                if (numero % 7 == 0 && numero % 10 == 3)
                {
                    Console.WriteLine("El numero es " + numero);
                }
            }


        }
    }
}
