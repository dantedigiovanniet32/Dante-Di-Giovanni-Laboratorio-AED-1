        static void EjercicioA()
        {
            int secreto = new Random().Next(1, 11);

            string lectura;
            int lecturaNumero = 0;

            Console.WriteLine("Adivina un numero entre 1 y 10");


            while (true)
            {
                Console.Write("Ingresa un numero: ");
                lectura = Console.ReadLine();


                bool valido = true;
                if (lectura.Length == 0) valido = false;
                for (int i = 0; i < lectura.Length && valido; i++)
                {
                    char c = lectura[i];
                    if (c < '0' || c > '9') valido = false;
                }

                if (valido)
                {
                    lecturaNumero = 0;
                    for (int i = 0; i < lectura.Length; i++)
                    {
                        lecturaNumero = lecturaNumero * 10 + (lectura[i] - '0');
                    }

                    if (lecturaNumero == secreto)
                    {
                        Console.WriteLine("Adivinaste!");
                        break;
                    }
                    else if (lecturaNumero > secreto)
                    {
                        Console.WriteLine("Es menor");
                    }
                    else
                    {
                        Console.WriteLine("Es mayor");
                    }

                }
                else
                {
                    Console.WriteLine("No es valido");
                }


            }
        }

        static void EjercicioD()
        {
            Console.Write("Elegi la operacion (suma, resta, multiplicacion, division): ");
            string operacion = Console.ReadLine();
            Console.Write("Ingresa el primer numero: ");
            int numero1 = Convert.ToInt32(Console.ReadLine());
            Console.Write("Ingresa el segundo numero: ");
            int numero2 = Convert.ToInt32(Console.ReadLine());
            double resultado = 0;
            string op = operacion.ToLower();
            if (op == "suma" || op == "+")
            {
                resultado = numero1 + numero2;
            }
            else if (op == "resta" || op == "-")
            {
                resultado = numero1 - numero2;
            }
            else if (op == "multiplicacion" || op == "*" || op == "multiplicar")
            {
                resultado = numero1 * numero2;
            }
            else if (op == "division" || op == "/")
            {
                resultado = (double)numero1 / numero2;
            }
            Console.WriteLine("");
            Console.WriteLine("El resutlado es: " + resultado);
        }

        static void EjercicioP()
        {
            int victoriasUsuario = 0;
            int victoriasComputadora = 0;

            string[] opciones = { "piedra", "papel", "tijera" };
            Random azar = new Random();

            for (int ronda = 1; ronda <= 3; ronda++)
            {
                Console.WriteLine("Escribi piedra, papel o tijera");
                string lectura = Console.ReadLine();


                string jugadaComputadora = opciones[azar.Next(0, 3)];

                if (lectura == jugadaComputadora)
                {
                    Console.WriteLine("Empate");
                }
                else if ((lectura == "piedra" && jugadaComputadora == "tijera") ||
                         (lectura == "papel" && jugadaComputadora == "piedra") ||
                         (lectura == "tijera" && jugadaComputadora == "papel"))
                {
                    victoriasUsuario++;
                    Console.WriteLine("Ganaste esta ronda");
                }
                else
                {
                    victoriasComputadora++;
                    Console.WriteLine("Perdiste esta ronda");
                }
            }

            if (victoriasUsuario > victoriasComputadora)
            {
                Console.WriteLine("Ganaste mas veces que la compuadora");
            }
            else if (victoriasUsuario < victoriasComputadora)
            {
                Console.WriteLine("La computadora gano mas veces");
            }
            else
            {
                Console.WriteLine("Empataron");
            }

        }
