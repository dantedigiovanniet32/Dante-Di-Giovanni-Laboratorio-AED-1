namespace ConsoleApp1
{
    internal class Program
    {
        static void Main(string[] args)
        {
            int[] numeros = new int[10];
            Random azar = new Random();
            int contador = 0;

            for (int i = 0; i < 10; i++)
            {
                numeros[i] = azar.Next(1, 101);


                if (numeros[i] > 50)
                {
                    contador++;
                }
            }

            Console.WriteLine("Hay " + contador + " numeros mayores a 50");





        }
    }
}
